# Forecast Agent 常用运行命令（PowerShell，旬度数据）
# 说明：
# - Excel 主时间列为 date（旬开始日，如 2016-03-01）；程序会派生 month=YYYY-MM-1/2/3
# - CLI 时间参数只接受旬开始日 YYYY-MM-DD，不接受 2025-06-3 这类内部旬标签
# - context_len 表示「旬」的条数（例如 90 旬 ≈ 30 个月）
# 当前项目若不是完整 git 仓库，uv run 触发 editable 构建时可能因 setuptools-scm 报错；
# 每次新开终端运行前可先设置下面的环境变量。

cd "E:\4.1\internship2\智能体框架\QuantaAlpha-main"

$env:SETUPTOOLS_SCM_PRETEND_VERSION_FOR_QUANTAALPHA = "0.1.0"

uv run python -m quantaalpha.forecast_agent.cli --model auto --province 河北 --as-of-month 2025-06-21 --test-start 2025-11-01 --test-end 2026-03-21 --context-len 270 --selection-metric mape --output-dir "forecast_agent_output/河北"


# ---- 显式指定 Excel 路径：TimesFM 搜索 context_len ----
uv run python -m quantaalpha.forecast_agent.cli --model timesfm --excel "data/processed_data/河北.xlsx" --as-of-month 2025-06-21 --test-start 2025-11-01 --test-end 2026-03-21 --timesfm-device cpu --selection-metric mape --output-dir "forecast_agent_output/河北"

# ---- 按省份自动查找 Excel ----
uv run python -m quantaalpha.forecast_agent.cli --model timesfm --province 河北 --as-of-month 2025-06-21 --test-start 2025-11-01 --test-end 2026-03-21 --timesfm-device cpu --selection-metric mape --output-dir "forecast_agent_output/河北"

# ---- SARIMAX（固定 context_len，单位：旬）----
uv run python -m quantaalpha.forecast_agent.cli --model sarimax --province 河北 --as-of-month 2025-06-21 --test-start 2025-11-01 --test-end 2026-03-21 --context-len 270 --selection-metric mape --output-dir "forecast_agent_output/河北"

# ---- Lasso ----
uv run python -m quantaalpha.forecast_agent.cli --model lasso --province 河北 --as-of-month 2025-06-21 --test-start 2025-11-01 --test-end 2026-03-21 --context-len 270 --selection-metric mape --output-dir "forecast_agent_output/河北"

# ---- 自动比选 ----
uv run python -m quantaalpha.forecast_agent.cli --model auto --candidate-models sarimax,lasso --province 河北 --as-of-month 2025-06-21 --test-start 2025-11-01 --test-end 2026-03-21 --context-len 270 --selection-metric mape --output-dir "forecast_agent_output/河北"

# =============================================================================
# 参数说明（旬度）
# =============================================================================
#
# --as-of-month
#   训练截止旬开始日，格式 YYYY-MM-DD；CLI 参数名未改，语义为旬
#
# --test-start / --test-end
#   测试旬区间；as_of 与 test_start 之间为 bridge，只预测不参与选参
#
# --context-len
#   历史窗口长度（旬数）。原月度 90 月约等于旬度 270 旬
#
# 特征说明：
# - 同比滞后为 Lag_36（12 月 × 3 旬/月）
# - LSTM 使用 Lag_36、HDD、is_heating_season
